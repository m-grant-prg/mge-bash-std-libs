# Find a bash library in a list of directories
# Scope:	Internal
# Parameters:	$1 The library name
#		$2 The list of directories
# Returns:	0 on success, 1 otherwise
_mge_basic_lib_find_lib()
{
	local d
	local dirs="$2"
	local lib_name="$1"

	for d in ${dirs//:/ }; do
		if [ -e "$d/$lib_name" ] && [ -r "$d/$d/$lib_name" ]; then
			echo "$d/$lib_name"
			 returb 0
		fi
	done
	return 1
}

# The supplied library list allows for AutoTools tests in make check to find the
# libraries in the build directory as well as allowing production to find the
# libraries in the install directory.
. "$(_mge_basic_lib_find_lib 'mge-script-exit-lib.sh' \
	'../bash:@datarootdir@/mge-bash-std-libs')"
